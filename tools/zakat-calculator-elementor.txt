<script>
function num(id){var v=parseFloat(document.getElementById(id).value);return isNaN(v)?0:v;}
function fmt(n){return 'PKR '+Math.round(n).toLocaleString('en-US');}
function calc(){
  var goldVal=num('goldTola')*num('goldRate'), silVal=num('silTola')*num('silRate');
  var assets=goldVal+silVal+num('cash')+num('bank')+num('invest')+num('stock')+num('loans');
  var liab=num('debts')+num('bills');
  var net=assets-liab;
  var type=document.querySelector('input[name=nisab]:checked').value;
  var nisab=type==='gold'?7.5*num('goldRate'):52.5*num('silRate');
  var due=nisab>0&&net>=nisab, zakat=due?net*0.025:0;
  document.getElementById('zakatAmt').textContent=fmt(zakat);
  document.getElementById('zakatMsg').textContent=due?'2.5% of your net zakatable wealth — may Allah accept it 🤲':(net<0?'Your liabilities exceed assets':'Your wealth is below nisab — no zakat due');
  document.getElementById('bdAssets').textContent=fmt(assets);
  document.getElementById('bdLiab').textContent=fmt(liab);
  document.getElementById('bdNet').textContent=fmt(net);
  document.getElementById('bdNisab').textContent=fmt(nisab);
}
document.querySelectorAll('input').forEach(function(el){el.addEventListener('input',calc);el.addEventListener('change',calc);});
calc();
</script>
