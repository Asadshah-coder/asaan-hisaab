<script>
var SYM={USD:'$',PKR:'Rs ',AED:'د.إ ',SAR:'﷼ '};
function rate(c){if(c==='USD')return 1;var v=parseFloat(document.getElementById('rt_'+c).value);return(v>0)?v:1;}
function fmt(n){return(Math.round(n*100)/100).toLocaleString('en-US');}
function swap(){var f=document.getElementById('from'),t=document.getElementById('to'),i=f.selectedIndex;f.selectedIndex=t.selectedIndex;t.selectedIndex=i;calc();}
function calc(){
var a=parseFloat(document.getElementById('amt').value)||0,
f=document.getElementById('from').value,t=document.getElementById('to').value;
var usd=a/rate(f), out=usd*rate(t);
document.getElementById('rVal').textContent=SYM[f]+fmt(a)+' = '+SYM[t]+fmt(out);
document.getElementById('rLbl').textContent=f+' → '+t;
document.getElementById('rRate').textContent='1 '+f+' = '+fmt(rate(t)/rate(f))+' '+t;
}
calc();
</script>
