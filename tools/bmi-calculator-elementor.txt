<script>
function cat(b){
if(b<18.5)return['Underweight','#3b82f6'];
if(b<25)return['Normal (Healthy)','#22c55e'];
if(b<30)return['Overweight','#f59e0b'];
return['Obese','#ef4444'];
}
function calc(){
var h=parseFloat(document.getElementById('h').value)||0,w=parseFloat(document.getElementById('w').value)||0;
var hu=document.getElementById('hu').value,wu=document.getElementById('wu').value;
var hm=hu==='cm'?h/100:h*0.3048, kg=wu==='kg'?w:w*0.45359237;
var bmi=(hm>0&&kg>0)?kg/(hm*hm):0;
var c=cat(bmi);
document.getElementById('rBmi').textContent=bmi.toFixed(1);
var rc=document.getElementById('rCat');rc.textContent=bmi>0?c[0]:'—';rc.style.color='#fff';
var pos=Math.min(100,Math.max(0,(bmi-14)/(40-14)*100));
document.getElementById('marker').style.left=pos+'%';
}
calc();
</script>
