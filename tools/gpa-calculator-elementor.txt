<script>
var GRADES=[["A+",4.0],["A",4.0],["A-",3.7],["B+",3.3],["B",3.0],["B-",2.7],["C+",2.3],["C",2.0],["C-",1.7],["D+",1.3],["D",1.0],["F",0.0]];
function gradeOpts(sel){return GRADES.map(function(g){return '<option value="'+g[1]+'"'+(g[0]===sel?' selected':'')+'>'+g[0]+' ('+g[1].toFixed(1)+')</option>';}).join('');}
function addRow(ch,gr){
var d=document.createElement('div');d.className='srow';
d.innerHTML='<input type="text" placeholder="Subject (optional)" style="width:100%;padding:9px 10px;border:1.5px solid #dde8e1;border-radius:9px;font-size:.95rem"><input type="number" class="ch" value="'+(ch||3)+'" min="0" step="0.5" oninput="calc()"><select class="gr" onchange="calc()">'+gradeOpts(gr||'A')+'</select><button class="del" onclick="this.parentElement.remove();calc()">×</button>';
document.getElementById('rows').appendChild(d);calc();
}
function calc(){
var ch=0,pts=0;
document.querySelectorAll('#rows .srow').forEach(function(r){
var c=parseFloat(r.querySelector('.ch').value)||0,g=parseFloat(r.querySelector('.gr').value)||0;
ch+=c;pts+=c*g;});
var gpa=ch>0?pts/ch:0;
document.getElementById('rGpa').textContent=gpa.toFixed(2);
document.getElementById('rCh').textContent=ch;
document.getElementById('rPts').textContent=(Math.round(pts*100)/100).toLocaleString('en-US');
}
function calcPct(){
var c=parseFloat(document.getElementById('cgpa').value)||0;
c=Math.min(4,Math.max(0,c));
document.getElementById('rPct').textContent=(c*25).toFixed(1)+'%';
}
addRow(3,'A');addRow(3,'B+');addRow(2,'A-');calcPct();
</script>
